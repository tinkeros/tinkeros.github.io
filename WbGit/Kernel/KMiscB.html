public U8 *UncachedDevCAlloc(I64 size)
{
  U8 *buf = CAllocAligned(size,16,Fs->code_heap);

  if (!buf) return NULL;

  return buf+dev.uncached_alias;
}

public U0 UncachedDevFree(U8 *d)
{
  if (d>dev.uncached_alias) Free(d-dev.uncached_alias);
  else Free(d);
}
